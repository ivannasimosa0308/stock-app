src/app/core/interceptors/auth.interceptor.ts

authInterceptor
Type : HttpInterceptorFn
Default value : ( req: HttpRequest<unknown>, next: HttpHandlerFn ): Observable<HttpEvent<unknown>> => { const authService = inject(AuthService); const router = inject(Router); // 1. Filtrado de dominio: Solo adjuntar token si la petición va dirigida a endpoints de la API de StockFlow const isTargetApi = req.url.startsWith(environment.supabaseUrl) || req.url.startsWith('/api'); let authReq = req; if (isTargetApi) { // Obtener sesión activa (Supabase o credenciales locales simuladas) const localSession = authService.readSessionFromStorage(); const token = localSession ? `mock-jwt-token-${localSession.role}` : null; if (token) { // HttpRequest es inmutable: se clona para inyectar cabecera Authorization authReq = req.clone({ setHeaders: { Authorization: `Bearer ${token}`, 'X-App-Client': 'StockFlow-Ionic-Client' } }); } } // 2. Flujo reactivo (Pipeline Onion) y captura centralizada de errores HTTP return next(authReq).pipe( catchError((error: HttpErrorResponse) => { if (error.status === 401) { // Token expirado o inválido: destruir sesión y forzar redirección con URL de retorno authService.logout(); router.navigate(['/login'], { queryParams: { returnUrl: router.url, sessionExpired: 'true' } }); } else if (error.status === 403) { // Intento de acceso sin privilegios de rol suficientes (RBAC) router.navigate(['/unauthorized']); } return throwError(() => error); }) ); }

Interceptor funcional para la inyección de token Bearer en peticiones al backend corporativo y gestión centralizada de excepciones de seguridad (401 Unauthorized / 403 Forbidden).

roleGuard
Type : any
Default value : ( allowedRoleOrRoute: 'admin' | 'user' | 'standard' | ActivatedRouteSnapshot = 'admin' ) => { // Caso 1: Invocado como factoría -> roleGuard('admin') if (typeof allowedRoleOrRoute === 'string') { const targetRole = allowedRoleOrRoute; return (route: ActivatedRouteSnapshot, state: RouterStateSnapshot): boolean | UrlTree => { const authService = inject(AuthService); const router = inject(Router); const currentUser = authService.currentUser(); const userRole = authService.userRole(); if (!authService.isAuthenticated() || !currentUser) { return router.createUrlTree(['/login'], { queryParams: { returnUrl: state.url } }); } const normalizedRole = userRole === 'standard' ? 'user' : userRole; if (normalizedRole === targetRole || userRole === 'admin') { return true; } // Registrar intento de intrusión para mostrar el aviso de Acceso Prohibido (403) sessionStorage.setItem( 'STOCKFLOW_FORBIDDEN_ATTEMPT', JSON.stringify({ attemptedUrl: state.url, user: currentUser.email || 'operador', role: normalizedRole, timestamp: new Date().toLocaleTimeString('es-ES') }) ); // Redirigir a /user-dashboard (o /unauthorized si se solicita explícitamente) return router.createUrlTree(['/user-dashboard'], { queryParams: { forbidden: 'true', attempted: state.url } }); }; } // Caso 2: Invocado directamente como CanActivateFn -> canActivate: [roleGuard] const route = allowedRoleOrRoute as ActivatedRouteSnapshot; const authService = inject(AuthService); const router = inject(Router); const currentUser = authService.currentUser(); const userRole = authService.userRole(); if (!authService.isAuthenticated() || !currentUser) { return router.createUrlTree(['/login']); } const allowedRoles = route?.data?.['roles'] as string[] | undefined; if (!allowedRoles || allowedRoles.length === 0) { return true; } if (allowedRoles.includes(userRole) || userRole === 'admin') { return true; } return router.createUrlTree(['/unauthorized']); }

Factoría de Guards que genera un CanActivateFn validando el rol exigido. Cumple con el Enunciado 4.2 del Dossier y registra alertas de intrusión (403).
